    <script>
      // ---------- Mock File Data ----------
      const sharedFiles = [
        { user: "Alice", name: "Biology_Notes.pdf", url: "#" },
        { user: "David", name: "History_Essay.docx", url: "#" },
        { user: "Sophia", name: "Math_Formulas.xlsx", url: "#" },
      ];

      const fileList = document.getElementById("fileList");
      const uploadForm = document.getElementById("uploadForm");
      const usernameInput = document.getElementById("username");
      const fileInput = document.getElementById("fileInput");

      // ---------- Function to Display Files ----------
      function renderFiles() {
        fileList.innerHTML = "";

        if (sharedFiles.length === 0) {
          fileList.innerHTML =
            '<p class="empty">No shared files yet. Upload one!</p>';
          return;
        }

        sharedFiles.forEach((file) => {
          const fileItem = document.createElement("div");
          fileItem.classList.add("file-item");

          fileItem.innerHTML = `
          <div class="file-info">
            <span>${file.user}</span>
            <small>${file.name}</small>
          </div>
          <a href="${file.url}" download>Download</a>
        `;
          fileList.appendChild(fileItem);
        });
      }

      // ---------- Handle Upload ----------
      uploadForm.addEventListener("submit", (e) => {
        e.preventDefault();

        const user = usernameInput.value.trim();
        const file = fileInput.files[0];

        if (!user || !file) {
          alert("Please enter your name and select a file.");
          return;
        }

        // In a real app, you'd send this to your backend using fetch() or axios
        // const formData = new FormData();
        // formData.append("user", user);
        // formData.append("file", file);
        // fetch("/upload", { method: "POST", body: formData });

        sharedFiles.push({
          user: user,
          name: file.name,
          url: "#", // would be the uploaded file link from backend
        });

        renderFiles();
        uploadForm.reset();
        alert("File shared successfully!");
      });

      // ---------- Initial Render ----------
      renderFiles();
    </script>
